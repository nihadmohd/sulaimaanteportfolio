    49→-- Auto-create a profile whenever Supabase Auth creates a user
    50→create or replace function public.handle_new_user()
    51→returns trigger
    52→language plpgsql security definer set search_path = public
    53→as $$
    54→begin
    55→  insert into public.profiles (id, email, full_name)
    56→  values (
    57→    new.id,
    58→    new.email,
    59→    coalesce(new.raw_user_meta_data ->> 'full_name', split_part(new.email, '@', 1))
    60→  )
    61→  on conflict (id) do nothing;
    62→  return new;
    63→end;
    64→$$;
    65→
    66→drop trigger if exists on_auth_user_created on auth.users;
    67→create trigger on_auth_user_created
    68→  after insert on auth.users
    69→  for each row execute function public.handle_new_user();

    83→-- role gate used by RLS policies (security definer avoids RLS recursion)
    84→create or replace function public.is_admin()
    85→returns boolean
    86→language sql stable security definer set search_path = public
    87→as $$
    88→  select coalesce(
    89→    exists (
    90→      select 1 from public.profiles p
    91→      where p.id = auth.uid() and p.role in ('editor','admin')
    92→    ), false
    93→  );
    94→$$;
    95→
    96→-- block role / activation self-escalation on profiles
    97→create or replace function public.guard_profile_mutations()
    98→returns trigger
    99→language plpgsql security definer set search_path = public
   100→as $$
   101→begin
   102→  if (new.role <> old.role or new.is_active <> old.is_active)
   103→     and not public.is_admin() then
   104→    raise exception 'SECURITY: role/is_active may only be changed by an admin';
   105→  end if;
   106→  return new;
   107→end;
   108→$$;
   109→
   110→drop trigger if exists trg_profiles_guard on public.profiles;
   111→create trigger trg_profiles_guard
   112→  before update on public.profiles
   113→  for each row execute function public.guard_profile_mutations();

   360→-- live counter (75s freshness window = 3 missed 25s heartbeats)
   361→create or replace view public.online_users_count as
   362→  select count(*)::int as online
   363→  from public.visitor_sessions
   364→  where last_seen_at > now() - interval '75 seconds';
   365→
   366→-- callable by anon/authenticated (REST fallback for the realtime counter)
   367→create or replace function public.get_online_count()
   368→returns integer
   369→language sql stable security definer set search_path = public
   370→as $$
   371→  select count(*)::int
   372→  from public.visitor_sessions
   373→  where last_seen_at > now() - interval '75 seconds';
   374→$$;
   375→
   376→-- offline sweep — schedule via pg_cron, or call from an edge function
   377→create or replace function public.sweep_presence()
   378→returns integer language plpgsql as $$
   379→declare n integer;
   380→begin
   381→  update public.visitor_sessions
   382→     set is_online = false
   383→   where is_online and last_seen_at < now() - interval '75 seconds';
   384→  get diagnostics n = row_count;
   385→  return n;
   386→end;
   387→$$;
