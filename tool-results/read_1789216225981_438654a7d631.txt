   157→   157→
   158→   158→/** ISO string → datetime-local input value ("YYYY-MM-DDTHH:mm"), local tz. */
   159→   159→function isoToLocalInput(iso: string | null): string {
   160→   160→  if (!iso) return "";
   161→   161→  try {
   162→   162→    const d = new Date(iso);
   163→   163→    if (Number.isNaN(d.getTime())) return "";
   164→   164→    const pad = (n: number) => String(n).padStart(2, "0");
   165→   165→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   166→   166→  } catch {
   167→   167→    return "";
   168→   168→  }
   169→   169→}
   170→   170→
   171→   171→/** datetime-local input value → ISO string, or null when empty. */
   172→   172→function localInputToIso(v: string): string | null {
   173→   173→  if (!v) return null;
   174→   174→  const t = new Date(v).getTime();
   175→   175→  return Number.isNaN(t) ? null : new Date(t).toISOString();
   176→   176→}
   177→   177→
   178→   178→/** Schedule chip: Scheduled (future start) · Live (in window) · Ended (past end). */
   179→   179→function scheduleChip(ad: AdDTO): { label: string; tone: Tone } {
   180→   180→  const now = Date.now();
   181→   181→  const start = ad.startAt ? new Date(ad.startAt).getTime() : null;
   182→   182→  const end = ad.endAt ? new Date(ad.endAt).getTime() : null;
   183→   183→  if (start != null && start > now) return { label: "Scheduled", tone: "amber" };
   184→   184→  if (end != null && end < now) return { label: "Ended", tone: "muted" };
   185→   185→  if (start != null || end != null) return { label: "Live", tone: "emerald" };
   186→   186→  return { label: "—", tone: "muted" };
   187→   187→}
   188→   188→
   189→   189→function ctrText(ad: AdDTO): string {
   190→   190→  if (!ad.impressions) return "—";
   191→   191→  return `${((ad.clicks / ad.impressions) * 100).toFixed(1)}%`;
   192→   192→}

   203→   203→    images: form.type === "marquee" ? form.images.slice(0, 16) : [],
   204→   204→    linkUrl: form.linkUrl.trim(),
   205→   205→    linkLabel: form.linkLabel.trim() || "Learn more",
   206→   206→    priority: Math.min(100, Math.max(0, Math.round(Number(form.priority) || 0))),
   207→   207→    startAt: localInputToIso(form.startAt),
   208→   208→    endAt: localInputToIso(form.endAt),
   209→   209→    active: form.active,
   210→   210→  };
   211→   211→}
