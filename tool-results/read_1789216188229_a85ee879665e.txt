   139→function formFromAd(ad: AdDTO): AdFormState {
   140→  return {
   141→    name: ad.name,
   142→    type: ad.type,
   143→    placement: ad.placement,
   144→    title: ad.title ?? "",
   145→    body: ad.body ?? "",
   146→    imageUrl: ad.imageUrl ?? "",
   147→    imageAlt: ad.imageAlt ?? "",
   148→    images: ad.images,
   149→    linkUrl: ad.linkUrl ?? "",
   150→    linkLabel: ad.linkLabel || "Learn more",
   151→    priority: String(ad.priority),
   152→    startAt: isoToLocalInput(ad.startAt),
   153→    endAt: isoToLocalInput(ad.endAt),
   154→    active: ad.active,
   155→  };
   156→}
   157→
   158→/** ISO string → datetime-local input value ("YYYY-MM-DDTHH:mm"), local tz. */
   159→function isoToLocalInput(iso: string | null): string {
   160→  if (!iso) return "";
   161→  try {
   162→    const d = new Date(iso);
   163→    if (Number.isNaN(d.getTime())) return "";
   164→    const pad = (n: number) => String(n).padStart(2, "0");
   165→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   166→  } catch {
   167→    return "";
   168→  }
   169→}
   170→
   171→/** datetime-local input value → ISO string, or null when empty. */
   172→function localInputToIso(v: string): string | null {
   173→  if (!v) return null;
   174→  const t = new Date(v).getTime();
   175→  return Number.isNaN(t) ? null : new Date(t).toISOString();
   176→}
   177→
   178→/** Schedule chip: Scheduled (future start) · Live (in window) · Ended (past end). */
   179→function scheduleChip(ad: AdDTO): { label: string; tone: Tone } {
   180→  const now = Date.now();
   181→  const start = ad.startAt ? new Date(ad.startAt).getTime() : null;
   182→  const end = ad.endAt ? new Date(ad.endAt).getTime() : null;
   183→  if (start != null && start > now) return { label: "Scheduled", tone: "amber" };
   184→  if (end != null && end < now) return { label: "Ended", tone: "muted" };
   185→  if (start != null || end != null) return { label: "Live", tone: "emerald" };
   186→  return { label: "—", tone: "muted" };
   187→}
   188→
   189→function ctrText(ad: AdDTO): string {
   190→  if (!ad.impressions) return "—";
   191→  return `${((ad.clicks / ad.impressions) * 100).toFixed(1)}%`;
   192→}

   194→function buildAdBody(form: AdFormState): Record<string, unknown> {
   195→  return {
   196→    name: form.name.trim(),
   197→    type: form.type,
   198→    placement: form.placement,
   199→    title: form.title.trim(),
   200→    body: form.body.trim(),
   201→    imageUrl: form.imageUrl.trim(),
   202→    imageAlt: form.imageAlt.trim(),
   203→    images: form.type === "marquee" ? form.images.slice(0, 16) : [],
   204→    linkUrl: form.linkUrl.trim(),
   205→    linkLabel: form.linkLabel.trim() || "Learn more",
   206→    priority: Math.min(100, Math.max(0, Math.round(Number(form.priority) || 0))),
   207→    startAt: localInputToIso(form.startAt),
   208→    endAt: localInputToIso(form.endAt),
   209→    active: form.active,
   210→  };
   211→}
