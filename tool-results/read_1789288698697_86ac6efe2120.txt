   114→/** Clipboard write with a legacy execCommand fallback (non-secure contexts). */
   115→async function copyText(text: string): Promise<boolean> {
   116→  try {
   117→    if (typeof navigator !== "undefined" && navigator.clipboard) {
   118→      await navigator.clipboard.writeText(text);
   119→      return true;
   120→    }
   121→  } catch {
   122→    /* fall through to the legacy path */
   123→  }
   124→  try {
   125→    const area = document.createElement("textarea");
   126→    area.value = text;
   127→    area.setAttribute("readonly", "");
   128→    area.style.position = "fixed";
   129→    area.style.opacity = "0";
   130→    document.body.appendChild(area);
   131→    area.select();
   132→    const ok = document.execCommand("copy");
   133→    area.remove();
   134→    return ok;
   135→  } catch {
   136→    return false;
   137→  }
   138→}

   140→function UtmBuilder() {
   141→  const [baseUrl, setBaseUrl] = React.useState(DEFAULT_BASE_URL);
   142→  const [values, setValues] = React.useState<UtmValues>({
   143→    source: "",
   144→    medium: "",
   145→    campaign: "",
   146→    term: "",
   147→    content: "",
   148→  });
   149→  const [copied, setCopied] = React.useState(false);
   150→
   151→  const campaignUrl = React.useMemo(() => {
   152→    const raw = baseUrl.trim();
   153→    if (!raw) return "";
   154→    const candidate = /^https?:\/\//i.test(raw) ? raw : `https://${raw}`;
   155→    let url: URL;
   156→    try {
   157→      url = new URL(candidate);
   158→    } catch {
   159→      return "";
   160→    }
   161→    const entries: Array<[string, string]> = [
   162→      ["utm_source", values.source.trim()],
   163→      ["utm_medium", values.medium.trim()],
   164→      ["utm_campaign", values.campaign.trim()],
   165→      ["utm_term", values.term.trim()],
   166→      ["utm_content", values.content.trim()],
   167→    ];
   168→    for (const [key, value] of entries) {
   169→      if (value) url.searchParams.set(key, value);
   170→    }
   171→    return url.toString();
   172→  }, [baseUrl, values]);
   173→
   174→  const trimmedBase = baseUrl.trim();
   175→  const paramCount = (["source", "medium", "campaign", "term", "content"] as const).filter(
   176→    (key) => values[key].trim()
   177→  ).length;
   178→  const missingRequired = !values.source.trim() || !values.campaign.trim();
   179→
   180→  const hint: { tone: "warn" | "ok"; text: string } = !trimmedBase || !campaignUrl
   181→    ? { tone: "warn", text: "Enter a valid base URL (e.g. https://mohdnihadkp.vercel.app) to build on." }
   182→    : missingRequired
   183→      ? { tone: "warn", text: "Add a campaign source and campaign name to complete the tag." }
   184→      : {
   185→          tone: "ok",
   186→          text: `${paramCount} UTM parameter${paramCount === 1 ? "" : "s"} tagged — ready to share.`,
   187→        };
   188→
   189→  const onCopy = async () => {
   190→    if (!campaignUrl) return;
   191→    const ok = await copyText(campaignUrl);
   192→    if (ok) {
   193→      setCopied(true);
   194→      toast({
   195→        title: "Campaign URL copied",
   196→        description: "Paste it straight into your campaign post or bio link.",
   197→      });
   198→      window.setTimeout(() => setCopied(false), 1800);
   199→    } else {
   200→      toast({
   201→        title: "Could not copy",
   202→        description: "Select the URL field and copy it manually.",
   203→        variant: "destructive",
   204→      });
   205→    }
   206→  };
   207→
   208→  const onOpen = () => {
   209→    if (campaignUrl) window.open(campaignUrl, "_blank", "noopener,noreferrer");
   210→  };
   211→
   212→  return (
   213→    <Card className="min-w-0 py-4 md:py-6">
   214→      <CardHeader className="px-4 pb-3 md:px-6">
   215→        <CardTitle className="flex items-center gap-2 text-base">
   216→          <Link2 className="size-4 text-gold" aria-hidden="true" />
   217→          UTM campaign builder
   218→        </CardTitle>
   219→        <CardDescription>
   220→          Tag any link with UTM parameters before sharing it — analytics-ready in one paste.
   221→        </CardDescription>
   222→      </CardHeader>
   223→      <CardContent className="px-4 pt-0 md:px-6">
   224→        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
   225→          <div className="space-y-1.5 sm:col-span-2 xl:col-span-3">
   226→            <Label htmlFor="utm-base" className="text-xs">
   227→              Base URL
   228→            </Label>
   229→            <Input
   230→              id="utm-base"
   231→              value={baseUrl}
   232→              onChange={(e) => setBaseUrl(e.target.value)}
   233→              placeholder={DEFAULT_BASE_URL}
   234→              inputMode="url"
   235→              className="h-9"
   236→            />
   237→          </div>
   238→          {UTM_FIELDS.map((field) => (
   239→            <div key={field.id} className="space-y-1.5">
   240→              <Label htmlFor={`utm-${field.id}`} className="text-xs">
   241→                {field.label}
   242→                {field.required ? (
   243→                  <>
   244→                    <span className="ml-0.5 text-gold" aria-hidden="true">
   245→                      *
   246→                    </span>
   247→                    <span className="sr-only">(required)</span>
   248→                  </>
   249→                ) : null}
   250→              </Label>
   251→              <Input
   252→                id={`utm-${field.id}`}
   253→                value={values[field.id]}
   254→                onChange={(e) =>
   255→                  setValues((v) => ({ ...v, [field.id]: e.target.value }) as UtmValues)
   256→                }
   257→                placeholder={field.placeholder}
   258→                className="h-9"
   259→              />
   260→            </div>
   261→          ))}
   262→        </div>
   263→
   264→        <div className="mt-4 rounded-lg border bg-muted/30 p-3">
   265→          <p className="text-[10px] font-medium uppercase tracking-[0.14em] text-muted-foreground">
   266→            Generated campaign URL
   267→          </p>
   268→          <div className="mt-2 flex flex-col gap-2 sm:flex-row sm:items-center">
   269→            <Input
   270→              readOnly
   271→              value={campaignUrl}
   272→              placeholder="https://…"
   273→              aria-label="Generated campaign URL"
   274→              className="h-9 min-w-0 flex-1 font-mono text-xs"
   275→            />
   276→            <div className="flex shrink-0 gap-2">
   277→              <Button
   278→                variant="outline"
   279→                size="sm"
   280→                className="h-9 gap-1.5"
   281→                onClick={onCopy}
   282→                disabled={!campaignUrl}
   283→              >
   284→                {copied ? (
   285→                  <Check className="size-3.5" aria-hidden="true" />
   286→                ) : (
   287→                  <Copy className="size-3.5" aria-hidden="true" />
   288→                )}
   289→                {copied ? "Copied" : "Copy"}
   290→              </Button>
   291→              <Button size="sm" className="h-9 gap-1.5" onClick={onOpen} disabled={!campaignUrl}>
   292→                <ExternalLink className="size-3.5" aria-hidden="true" />
   293→                Open
   294→              </Button>
   295→            </div>
   296→          </div>
   297→          <p
   298→            className={cn(
   299→              "mt-2 flex items-center gap-1.5 text-xs",
   300→              hint.tone === "warn" ? "text-amber-600 dark:text-amber-400" : "text-muted-foreground"
   301→            )}
   302→          >
   303→            {hint.tone === "warn" ? (
   304→              <AlertCircle className="size-3.5 shrink-0" aria-hidden="true" />
   305→            ) : (
   306→              <Check
   307→                className="size-3.5 shrink-0 text-emerald-600 dark:text-emerald-400"
   308→                aria-hidden="true"
   309→              />
   310→            )}
   311→            {hint.text}
   312→          </p>
   313→        </div>
   314→      </CardContent>
   315→    </Card>
   316→  );
   317→}

   331→        {
   332→          label: "Meta description (70+ chars)",
   333→          detail: `${seo.defaultDescription.trim().length} chars`,
   334→          passing: seo.defaultDescription.trim().length >= 70,

   337→          label: "Target keywords (3+)",
   338→          detail: `${seo.keywords.length} keywords`,
