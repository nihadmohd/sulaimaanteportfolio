   146→
   147→function asString(v: unknown, fallback = ""): string {
   148→  return typeof v === "string" ? v : fallback;
   149→}
   150→function asBool(v: unknown, fallback = false): boolean {
   151→  return typeof v === "boolean" ? v : fallback;
   152→}
   153→function asStrArr(v: unknown): string[] {
   154→  return Array.isArray(v) ? v.filter((x): x is string => typeof x === "string") : [];
   155→}
   156→function asRecord(v: unknown): Record<string, string> {
   157→  if (typeof v !== "object" || v === null || Array.isArray(v)) return {};
   158→  const out: Record<string, string> = {};
   159→  for (const [k, val] of Object.entries(v as Record<string, unknown>)) {
   160→    if (typeof val === "string") out[k] = val;
   161→  }
   162→  return out;
   163→}
