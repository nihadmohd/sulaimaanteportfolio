   139→function VentureCard({ venture }: { venture: VentureDTO }) {
   140→  const meta = CATEGORY_META[venture.category] ?? CATEGORY_META.venture;

   142→  const visibleHighlights = venture.highlights.slice(0, 4);
   143→  const extraHighlights = venture.highlights.length - visibleHighlights.length;
   144→
   145→  return (
   146→    <article className="flex h-full flex-col rounded-xl border bg-card p-4 shadow-xs transition-shadow hover:shadow-md sm:p-5">
   147→      {/* cover image or copper category medallion */}
   148→      {venture.imageUrl ? (
   149→        <img
   150→          src={venture.imageUrl}
   151→          alt={`${venture.name} cover`}
   152→          loading="lazy"
   153→          className="aspect-video w-full rounded-lg border object-cover"
   154→        />
   155→      ) : (
   156→        <div className="flex aspect-video w-full items-center justify-center rounded-lg border border-gold/25 bg-gold/[0.05]">
   157→          <span
   158→            aria-hidden="true"
   159→            className="flex size-12 items-center justify-center rounded-lg border border-gold/30 bg-gold/10 text-gold"
   160→          >
   161→            <meta.icon className="size-6" strokeWidth={1.75} />
   162→          </span>
   163→        </div>
   164→      )}
   165→
   166→      <div className="mt-3.5 flex flex-wrap items-center gap-1.5">
   167→        <Badge
   168→          variant="outline"
   169→          className={cn("gap-1.5 px-2 py-0 text-[10px] font-medium capitalize", status.badge)}
   170→        >
   171→          <span aria-hidden="true" className={cn("size-1.5 rounded-full", status.dot)} />
   172→          {status.label}
   173→        </Badge>
   174→        <Badge variant="outline" className="px-2 py-0 text-[10px] font-medium capitalize text-muted-foreground">
   175→          {meta.label}
   176→        </Badge>
   177→        {venture.isFeatured ? (
   178→          <Badge variant="outline" className="border-gold/40 px-2 py-0 text-[10px] font-medium text-gold">
   179→            Featured
   180→          </Badge>
   181→        ) : null}
   182→      </div>
   183→
   184→      <h3 className="mt-2.5 text-base font-semibold tracking-tight md:text-lg">
   185→        {venture.name}
   186→      </h3>
   187→      {venture.location ? (
   188→        <p className="mt-1 inline-flex items-center gap-1.5 text-xs text-muted-foreground">
   189→          <MapPin className="size-3.5 shrink-0 text-gold/80" aria-hidden="true" />
   190→          {venture.location}
   191→        </p>
   192→      ) : null}
   193→      {venture.tagline ? (
   194→        <p className="mt-2 text-[13px] leading-relaxed text-muted-foreground md:text-sm">
   195→          {venture.tagline}
   196→        </p>
   197→      ) : null}
   198→
   199→      {visibleHighlights.length > 0 ? (
   200→        <ul className="mt-3.5 space-y-2">
   201→          {visibleHighlights.map((item) => (
   202→            <li key={item} className="flex items-start gap-2.5 text-[13px] leading-relaxed">
   203→              <span
   204→                aria-hidden="true"
   205→                className="mt-0.5 flex size-4 shrink-0 items-center justify-center rounded-full border border-primary/30 bg-primary/10 text-primary"
   206→              >
   207→                <Check className="size-2.5" strokeWidth={2.5} />
   208→              </span>
   209→              <span className="text-muted-foreground">{item}</span>
   210→            </li>
   211→          ))}
   212→          {extraHighlights > 0 ? (
   213→            <li className="pl-[26px] text-xs text-muted-foreground">+{extraHighlights} more</li>
   214→          ) : null}
   215→        </ul>
   216→      ) : null}
   217→
   218→      {venture.collabRoles.length > 0 ? (
   219→        <div className="mt-3.5">
   220→          <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-muted-foreground">
   221→            Looking for
   222→          </p>
   223→          <div className="mt-1.5 flex flex-wrap gap-1.5">
   224→            {venture.collabRoles.map((role) => (
   225→              <Badge
   226→                key={role}
   227→                variant="outline"
   228→                className="border-gold/40 bg-gold/[0.06] px-2 py-0 text-[10px] font-medium text-gold"
   229→              >
   230→                {role}
   231→              </Badge>
   232→            ))}
   233→          </div>
   234→        </div>
   235→      ) : null}
   236→
   237→      {venture.websiteUrl ? (
   238→        <div className="mt-auto pt-4">
   239→          <a
   240→            href={venture.websiteUrl}
   241→            target="_blank"
   242→            rel="noopener noreferrer"
   243→            className="inline-flex h-9 items-center gap-2 rounded-md border px-3 text-sm font-medium transition-colors hover:bg-muted focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   244→            aria-label={`Visit the ${venture.name} website`}
   245→          >
   246→            <ExternalLink className="size-4" aria-hidden="true" />
   247→            Visit website
   248→          </a>
   249→        </div>
   250→      ) : null}
   251→    </article>
   252→  );
   253→}

   257→/* ------------------------------------------------------------------ */
   258→
   259→export default function VenturesView() {
   260→  const [submitted, setSubmitted] = React.useState<InquiryDTO | null>(null);
   261→  const [submitError, setSubmitError] = React.useState<string | null>(null);
   262→
   263→  const venturesQuery = useQuery({
   264→    queryKey: ["ventures"],
   265→    queryFn: () => apiFetch<VenturesResponse>("/api/ventures"),
   266→    staleTime: 60_000,
   267→    retry: 1,
   268→  });
   269→
   270→  const ventures = venturesQuery.data?.items ?? [];
   271→
   272→  const jsonLd = React.useMemo(() => {
   273→    if (ventures.length === 0) return undefined;
   274→    return {
   275→      "@context": "https://schema.org",
   276→      "@type": "ItemList",
   277→      name: "MN.KP ventures & business ideas",
   278→      itemListElement: ventures.map((venture, index) => ({
   279→        "@type": "ListItem",
   280→        position: index + 1,
   281→        item: {
   282→          "@type": "Organization",
   283→          name: venture.name,
   284→          description: venture.tagline ?? undefined,
   285→          url: venture.websiteUrl ?? `${SITE.url}/ventures`,
   286→          ...(venture.location
   287→            ? {
   288→                location: {
   289→                  "@type": "Place",
   290→                  name: venture.location,
   291→                },
   292→              }
   293→            : {}),
   294→        },
   295→      })),
   296→    };
   297→  }, [ventures]);
   298→
   299→  const {
   300→    register,
   301→    handleSubmit,
   302→    reset,
   303→    setValue,
   304→    control,
   305→    formState: { errors, isSubmitting },
   306→  } = useForm<PitchFormValues>({
   307→    resolver: zodResolver(pitchFormSchema),
   308→    defaultValues: {
   309→      intent: "pitch",
   310→      venture: NEW_IDEA,
   311→      name: "",
   312→      email: "",
   313→      phone: "",
   314→      message: "",
   315→    },
   316→  });
   317→
   318→  // Radix Selects mirror their value into local state + the RHF fields.
   319→  const [intentValue, setIntentValue] = React.useState<IntentValue>("pitch");
   320→  const [ventureValue, setVentureValue] = React.useState<string>(NEW_IDEA);
   321→
   322→  const submitMutation = useMutation({
   323→    mutationFn: (values: PitchFormValues) => {
   324→      const intentLabel =
   325→        INTENT_OPTIONS.find((o) => o.value === values.intent)?.label ?? "Startup idea";
   326→      const ventureName =
   327→        values.venture === NEW_IDEA
   328→          ? "A new idea"
   329→          : (ventures.find((v) => v.id === values.venture)?.name ?? "A new idea");
   330→      return apiFetch<InquiryDTO>("/api/inquiries", {
   331→        method: "POST",
   332→        body: JSON.stringify({
   333→          type: "venture",
   334→          subject: `${intentLabel} — ${ventureName}`,
   335→          name: values.name,
   336→          email: values.email,
   337→          phone: values.phone,
   338→          message: values.message,
   339→        }),
   340→      });
   341→    },
   342→    onSuccess: (inquiry) => {
   343→      setSubmitted(inquiry);
   344→      setSubmitError(null);
   345→      reset({ intent: "pitch", venture: NEW_IDEA, name: "", email: "", phone: "", message: "" });
   346→      setIntentValue("pitch");
   347→      setVentureValue(NEW_IDEA);
   348→      toast({
   349→        title: "Pitch received",
   350→        description: "Your startup idea landed — expect a reply within 24 hours.",
   351→      });
   352→    },
   353→    onError: (e: Error) => {
   354→      setSubmitError(e.message);
   355→      toast({
   356→        title: "Could not send your pitch",
   357→        description: e.message,
   358→        variant: "destructive",
   359→      });
   360→    },
   361→  });
   362→
   363→  const onSubmit = (values: PitchFormValues) => {
   364→    setSubmitError(null);
   365→    void submitMutation.mutateAsync(values);
   366→  };
   367→
   368→  const sendAnother = () => {
   369→    setSubmitted(null);
   370→    setSubmitError(null);
   371→    setIntentValue("pitch");
   372→    setVentureValue(NEW_IDEA);
   373→  };
   374→
   375→  const scrollToForm = () => {
   376→    document.getElementById("pitch-your-idea")?.scrollIntoView({ behavior: "smooth", block: "start" });
   377→  };
   378→
   379→  // useWatch (single call, whole values) instead of form.watch — satisfies
   380→  // react-hooks/incompatible-library; defaults are complete so the cast is safe.
   381→  const watched = useWatch({ control }) as PitchFormValues;
   382→  const currentIntent = watched.intent;
   383→
   384→  return (
   385→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:px-6 md:py-14 lg:px-8">
   386→      <SEOHead
   387→        title="Ventures & Business Ideas — Calicut Store, Chaliyam Connect & More | MN.KP"
   388→        description="Every MN.KP business idea and venture in one place — the Calicut Store, Chaliyam Connect, digital products and long-horizon startup ideas from Calicut, Kerala. Got a startup idea? Join me and build it together."
   389→        canonicalPath="/ventures"
   390→        ogImage="/images/brand/og-cover.webp"
   391→        jsonLd={jsonLd}
   392→      />
   393→
   394→      <Breadcrumbs items={[{ label: "Home", href: "#/" }, { label: "Ventures" }]} />
   395→
   396→      <header className="mt-5 md:mt-8">
   397→        <SectionHeading
   398→          microLabel="Ventures & business ideas"
   399→          title="Everything I'm building — on one page"
   400→          description="From a curated Calicut store to community initiatives and long-horizon startup ideas, these are the ventures I'm actively growing. If one of them speaks to you, there's a seat at the table."
   401→        />
   402→      </header>
   403→
   404→      {/* ventures grid */}
   405→      <div className="mt-8 md:mt-12">
   406→        <DataState query={venturesQuery} skeletonRows={6}>
   407→          {(data) => (
   408→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 sm:gap-5 lg:grid-cols-3 lg:gap-6">
   409→              {data.items.map((venture) => (
   410→                <VentureCard key={venture.id} venture={venture} />
   411→              ))}
   412→            </div>
   413→          )}
   414→        </DataState>
   415→      </div>
   416→
   417→      {/* join me band */}
   418→      <section
   419→        className="mt-10 rounded-xl border border-gold/40 bg-gold/[0.05] p-5 md:mt-16 md:p-8"
   420→        aria-labelledby="join-me-heading"
   421→      >
   422→        <p className="text-xs font-medium uppercase tracking-[0.2em] text-gold">Join me</p>
   423→        <h2 id="join-me-heading" className="mt-2 text-balance text-xl font-semibold tracking-tight md:text-2xl lg:text-3xl">
   424→          Build the next startup with me
   425→        </h2>
   426→        <p className="mt-2.5 max-w-2xl text-pretty text-[13px] leading-relaxed text-muted-foreground md:mt-3 md:text-sm lg:text-base">
   427→          Every venture above started as a conversation. I partner with operators, investors,
   428→          creators and local businesses — whether you want to co-found something new, join a
   429→          live venture, or pitch an idea of your own. Current focus:{" "}
   430→          <span className="font-medium text-foreground">
   431→            {ventures[0]?.name ?? "Calicut Store"} collabs
   432→          </span>
   433→          .
   434→        </p>
   435→        <div className="mt-5 flex flex-wrap items-center gap-2.5 md:mt-6 md:gap-3">
   436→          <Button size="lg" className="h-10 gap-2 md:h-11" onClick={scrollToForm}>
   437→            <Lightbulb className="size-4" aria-hidden="true" />
   438→            Pitch your idea
   439→          </Button>
   440→          <ALink href={SITE.whatsappUrl}>
   441→            <Button
   442→              size="lg"
   443→              variant="outline"
   444→              className="h-10 gap-2 border-gold/50 text-gold hover:bg-gold/10 md:h-11"
   445→            >
   446→              <MessageCircle className="size-4" aria-hidden="true" />
   447→              WhatsApp me
   448→            </Button>
   449→          </ALink>
   450→        </div>
   451→        <p className="mt-4 flex items-center gap-1.5 text-xs text-muted-foreground">
   452→          <Users className="size-3.5 text-gold/80" aria-hidden="true" />
   453→          Collaboration is welcome at every stage — idea, incubation or live.
   454→        </p>
   455→      </section>
   456→
   457→      {/* startup-idea inquiry form */}
   458→      <section id="pitch-your-idea" className="mt-10 scroll-mt-24 md:mt-14" aria-labelledby="pitch-form-heading">
   459→        <SectionHeading
   460→          microLabel="Startup inquiry"
   461→          title="Tell me about your idea"
   462→          description="Two minutes, no pitch deck needed. Serious ideas get a personal reply within 24 hours — usually much sooner over WhatsApp."
   463→        />
   464→        <div className="mt-5 md:mt-8">
   465→          {submitted ? (
   466→            <div className="rounded-xl border bg-card p-2 shadow-xs">
   467→              <SuccessState
   468→                title="Pitch received"
   469→                description={
   470→                  <>
   471→                    Your reference number is{" "}
   472→                    <span className="font-semibold tabular-nums text-foreground">
   473→                      {submitted.id.slice(0, 8).toUpperCase()}
   474→                    </span>
   475→                    . It lands in the MN.KP inbox as a venture inquiry — expect a response
   476→                    within 24 hours.
   477→                  </>
   478→                }
   479→                action={
   480→                  <Button onClick={sendAnother} variant="outline" className="gap-2">
   481→                    <RotateCcw className="size-4" aria-hidden="true" />
   482→                    Send another idea
   483→                  </Button>
   484→                }
   485→              />
   486→            </div>
   487→          ) : submitError ? (
   488→            <div className="rounded-xl border bg-card p-2 shadow-xs">
   489→              <ErrorState
   490→                message={submitError}
   491→                onRetry={() => {
   492→                  setSubmitError(null);
   493→                }}
   494→              />
   495→            </div>
   496→          ) : (
   497→            <form
   498→              onSubmit={handleSubmit(onSubmit)}
   499→              noValidate
   500→              className="rounded-2xl border bg-card p-4 shadow-xs sm:p-6 md:p-8"
   501→            >
   502→              <p className="inline-flex items-center gap-1.5 text-xs font-medium uppercase tracking-[0.2em] text-gold">
   503→                <Handshake className="size-3.5" aria-hidden="true" />
   504→                Join me — startup idea form
   505→              </p>
   506→              <div className="gold-rule mt-3 mb-5 w-16 md:mb-6" aria-hidden="true" />
   507→
   508→              <div className="grid grid-cols-1 gap-5 sm:grid-cols-2">
   509→                <div className="space-y-2">
   510→                  <Label htmlFor="pitch-intent">I want to</Label>
   511→                  <Select
   512→                    value={intentValue}
   513→                    onValueChange={(value) => {
   514→                      const next = value as IntentValue;
   515→                      setIntentValue(next);
   516→                      setValue("intent", next, { shouldValidate: true });
   517→                    }}
   518→                  >
   519→                    <SelectTrigger id="pitch-intent" aria-invalid={errors.intent ? true : undefined}>
   520→                      <SelectValue placeholder="Choose an intent" />
   521→                    </SelectTrigger>
   522→                    <SelectContent>
   523→                      {INTENT_OPTIONS.map((option) => (
   524→                        <SelectItem key={option.value} value={option.value}>
   525→                          {option.label}
   526→                        </SelectItem>
   527→                      ))}
   528→                    </SelectContent>
   529→                  </Select>
   530→                  {errors.intent ? (
   531→                    <p role="alert" className="text-xs text-destructive">
   532→                      {errors.intent.message}
   533→                    </p>
   534→                  ) : null}
   535→                </div>
   536→
   537→                <div className="space-y-2">
   538→                  <Label htmlFor="pitch-venture">The venture</Label>
   539→                  <Select
   540→                    value={ventureValue}
   541→                    onValueChange={(value) => {
   542→                      setVentureValue(value);
   543→                      setValue("venture", value, { shouldValidate: true });
   544→                    }}
   545→                  >
   546→                    <SelectTrigger id="pitch-venture" aria-invalid={errors.venture ? true : undefined}>
   547→                      <SelectValue placeholder="Pick a venture" />
   548→                    </SelectTrigger>
