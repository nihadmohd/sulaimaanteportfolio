   127→function asString(v: unknown, fallback = ""): string {
   128→  return typeof v === "string" ? v : fallback;
   129→}
   130→function asBool(v: unknown, fallback = false): boolean {
   131→  return typeof v === "boolean" ? v : fallback;
   132→}
   133→function asStrArr(v: unknown): string[] {
   134→  return Array.isArray(v) ? v.filter((x): x is string => typeof x === "string") : [];
   135→}
   136→function asRecord(v: unknown): Record<string, string> {
   137→  if (typeof v !== "object" || v === null || Array.isArray(v)) return {};
   138→  const out: Record<string, string> = {};
   139→  for (const [k, val] of Object.entries(v as Record<string, unknown>)) {
   140→    if (typeof val === "string") out[k] = val;
   141→  }
   142→  return out;
   143→}
