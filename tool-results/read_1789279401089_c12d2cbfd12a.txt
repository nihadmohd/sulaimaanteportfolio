    77→

    88→
    89→const productFormSchema = z.object({
    90→  name: z.string().min(3, "Product name is required").max(140),
    91→  slug: z.string().max(96).optional(),
    92→  tagline: z.string().max(160).optional(),
    93→  description: z.string().optional(),
    94→  brand: z.string().max(60).optional(),
    95→  merchant: z.string().max(60).optional(),
    96→  imageUrl: z.string().optional(),
    97→  affiliateUrl: z.string().min(3, "Affiliate URL is required").max(600),
    98→  price: z
    99→    .string()
   100→    .optional()
   101→    .refine((v) => !v || (!Number.isNaN(Number(v)) && Number(v) >= 0), "Enter a valid price"),
   102→  compareAtPrice: z
   103→    .string()
   104→    .optional()
   105→    .refine((v) => !v || (!Number.isNaN(Number(v)) && Number(v) >= 0), "Enter a valid price"),
   106→  reviewCount: z
   107→    .string()
   108→    .optional()
   109→    .refine((v) => !v || (Number.isInteger(Number(v)) && Number(v) >= 0), "Whole number"),
   110→  status: z.enum(["active", "draft", "archived"]),
   111→  isFeatured: z.boolean(),
   112→  categoryId: z.string().optional(),
   113→});
   114→
   115→type ProductFormValues = z.infer<typeof productFormSchema>;
   116→
   117→interface SpecRow {
   118→  key: string;
   119→  value: string;
   120→}
   121→
   122→/** Snapshot payload for the local crash-recovery autosave. */
   123→interface ProductDraftData {
   124→  values: ProductFormValues;
   125→  gallery: string[];
   126→  pros: string[];
   127→  cons: string[];
   128→  keySpecs: SpecRow[];
   129→  rating: number;
   130→}
   131→
   132→export default function ProductEditView() {
   133→  const { id } = useHashParams<{ id: string }>();
   134→  const isNew = !id || id === "new";
   135→  const { isLoading, allowed } = useAdminGuard();
   136→  const queryClient = useQueryClient();
   137→  const categories = useCategories("store");
   138→
   139→  const [gallery, setGallery] = React.useState<string[]>([]);
   140→  const [pros, setPros] = React.useState<string[]>([]);
   141→  const [cons, setCons] = React.useState<string[]>([]);
   142→  const [keySpecs, setKeySpecs] = React.useState<SpecRow[]>([]);
   143→  const [rating, setRating] = React.useState(0);
   144→  const [slugTouched, setSlugTouched] = React.useState(false);
   145→  const [galleryInput, setGalleryInput] = React.useState("");
   146→  const [descTab, setDescTab] = React.useState<"write" | "preview">("write");
   147→  const [extraDirty, setExtraDirty] = React.useState(false);
   148→
   149→  const form = useForm<ProductFormValues>({
   150→    resolver: zodResolver(productFormSchema),
   151→    defaultValues: {
   152→      name: "",
   153→      slug: "",
   154→      tagline: "",
   155→      description: "",
   156→      brand: "",
   157→      merchant: "",
   158→      imageUrl: "",
   159→      affiliateUrl: "",
   160→      price: "",
   161→      compareAtPrice: "",
   162→      reviewCount: "",
   163→      status: "active",
   164→      isFeatured: false,
   165→      categoryId: "",
   166→    },
   167→  });
   168→
   169→  const productQuery = useQuery({
   170→    queryKey: ["admin-product", id],
   171→    queryFn: () => apiFetch<ProductDTO>(`/api/products/${id}`),
   172→    enabled: allowed && !isNew,
   173→    retry: 1,
   174→  });
   175→
   176→  React.useEffect(() => {
   177→    if (!productQuery.data) return;
   178→    const p = productQuery.data;
   179→    form.reset({
   180→      name: p.name,
   181→      slug: p.slug,
   182→      tagline: p.tagline ?? "",
   183→      description: p.description ?? "",
   184→      brand: p.brand ?? "",
   185→      merchant: p.merchant ?? "",
   186→      imageUrl: p.imageUrl ?? "",
   187→      affiliateUrl: p.affiliateUrl,
   188→      price: p.price != null ? String(p.price) : "",
   189→      compareAtPrice: p.compareAtPrice != null ? String(p.compareAtPrice) : "",
   190→      reviewCount: String(p.reviewCount ?? 0),
   191→      status: p.status,
   192→      isFeatured: p.isFeatured,
   193→      categoryId: p.category?.id ?? "",
   194→    });
   195→    setGallery(p.gallery);
   196→    setPros(p.pros);
   197→    setCons(p.cons);
   198→    setKeySpecs(Object.entries(p.keySpecs).map(([key, value]) => ({ key, value })));
   199→    setRating(p.rating);
   200→    setSlugTouched(true);
   201→    setExtraDirty(false);
   202→  }, [productQuery.data, form]);
   203→
   204→  // useWatch (single call, whole values) instead of form.watch — satisfies
   205→  // react-hooks/incompatible-library; defaults are complete so the cast is safe.
   206→  const watched = useWatch({ control: form.control }) as ProductFormValues;
   207→
   208→  const effectiveSlug = slugTouched && watched.slug ? slugify(watched.slug) : slugify(watched.name);
   209→  const priceNum = watched.price ? Number(watched.price) : 0;
   210→  const compareNum = watched.compareAtPrice ? Number(watched.compareAtPrice) : 0;
   211→  const discount =
   212→    compareNum > priceNum && priceNum > 0 ? Math.round((1 - priceNum / compareNum) * 100) : 0;
   213→  const isDirty = form.formState.isDirty || extraDirty;
   214→
   215→  /* ---------------- local autosave + crash recovery ---------------- */
   216→
   217→  const draftKey = `mnkp_draft_product_${isNew ? "new" : id}`;
   218→
   219→  const draft = useDraftAutosave<ProductDraftData>({
   220→    key: draftKey,
   221→    ready: isNew || !!productQuery.data,
   222→    isDirty,
   223→    tick: JSON.stringify([watched, gallery, pros, cons, keySpecs, rating]),
   224→    capture: () => ({ values: form.getValues(), gallery, pros, cons, keySpecs, rating }),
   225→    differs: (snap) => {
   226→      const data = snap.data;
   227→      if (!data || typeof data !== "object" || typeof data.values !== "object" || !data.values) {
   228→        return false; // unrecognized snapshot shape
   229→      }
   230→      if (isNew) {
   231→        return !!(data.values.name?.trim() || data.values.description?.trim());
   232→      }
   233→      const p = productQuery.data;
   234→      if (!p) return false;
   235→      // only offer recovery when the local copy is newer than the server copy
   236→      const serverTime = new Date(p.updatedAt).getTime();
   237→      if (Number.isFinite(serverTime) && snap.savedAt <= serverTime) return false;
   238→      return (
   239→        data.values.name !== p.name ||
   240→        (data.values.description ?? "") !== (p.description ?? "") ||
   241→        data.values.status !== p.status ||
   242→        JSON.stringify(data.pros ?? []) !== JSON.stringify(p.pros) ||
   243→        JSON.stringify(data.cons ?? []) !== JSON.stringify(p.cons)
   244→      );
   245→    },
   246→    onRestore: (data) => {
   247→      const v = data.values;
   248→      form.reset({
   249→        name: v?.name ?? "",
   250→        slug: v?.slug ?? "",
   251→        tagline: v?.tagline ?? "",
   252→        description: v?.description ?? "",
   253→        brand: v?.brand ?? "",
   254→        merchant: v?.merchant ?? "",
   255→        imageUrl: v?.imageUrl ?? "",
   256→        affiliateUrl: v?.affiliateUrl ?? "",
   257→        price: v?.price ?? "",
   258→        compareAtPrice: v?.compareAtPrice ?? "",
   259→        reviewCount: v?.reviewCount ?? "",
   260→        status: v?.status ?? "active",
   261→        isFeatured: !!v?.isFeatured,
   262→        categoryId: v?.categoryId ?? "",
   263→      });
   264→      setGallery(Array.isArray(data.gallery) ? data.gallery : []);
   265→      setPros(Array.isArray(data.pros) ? data.pros : []);
   266→      setCons(Array.isArray(data.cons) ? data.cons : []);
   267→      setKeySpecs(Array.isArray(data.keySpecs) ? data.keySpecs : []);
   268→      setRating(typeof data.rating === "number" ? data.rating : 0);
   269→      setSlugTouched(true);
   270→      setExtraDirty(true);
   271→      toast({ title: "Local draft restored", description: "Review it and save when you are ready." });
   272→    },
   273→  });
   274→
   275→  const saveMutation = useMutation({
   276→    mutationFn: (values: ProductFormValues) => {
   277→      const body: Record<string, unknown> = {
   278→        name: values.name,
   279→        affiliateUrl: values.affiliateUrl,
   280→        status: values.status,
   281→        isFeatured: values.isFeatured,
   282→        currency: "INR",
   283→        imageUrl: values.imageUrl || "",
   284→        gallery: gallery.filter(Boolean),
   285→        pros: pros.map((s) => s.trim()).filter(Boolean),
   286→        cons: cons.map((s) => s.trim()).filter(Boolean),
   287→        keySpecs: Object.fromEntries(
   288→          keySpecs.filter((row) => row.key.trim()).map((row) => [row.key.trim(), row.value])
   289→        ),
   290→        rating,
   291→        reviewCount: Number(values.reviewCount || 0),
   292→        price: values.price ? Number(values.price) : null,
   293→        compareAtPrice: values.compareAtPrice ? Number(values.compareAtPrice) : null,
   294→        categoryId: values.categoryId || null,
   295→        tagline: values.tagline || undefined,
   296→        description: values.description || undefined,
   297→        brand: values.brand || undefined,
   298→        merchant: values.merchant || undefined,
   299→      };
   300→      if (slugTouched && values.slug) body.slug = slugify(values.slug);
   301→      return isNew
   302→        ? apiFetch<ProductDTO>("/api/products", { method: "POST", body: JSON.stringify(body) })
   303→        : apiFetch<ProductDTO>(`/api/products/${id}`, { method: "PATCH", body: JSON.stringify(body) });
   304→    },
   305→    onSuccess: (product) => {
   306→      toast({ title: isNew ? "Product created" : "Product saved", description: product.name });
   307→      setExtraDirty(false);
   308→      draft.clear();
   309→      void queryClient.invalidateQueries({ queryKey: ["admin-products"] });
   310→      void queryClient.invalidateQueries({ queryKey: ["admin-stats"] });
   311→      if (isNew) navigate(`/admin/products/${product.id}`);
   312→      else form.reset(form.getValues());
   313→    },
   314→    onError: (e: Error) =>
   315→      toast({ title: "Save failed", description: e.message, variant: "destructive" }),
   316→  });
   317→
   318→  const deleteMutation = useMutation({
   319→    mutationFn: () => apiFetch(`/api/products/${id}`, { method: "DELETE" }),
   320→    onSuccess: () => {
   321→      toast({ title: "Product deleted" });
   322→      draft.clear();
   323→      void queryClient.invalidateQueries({ queryKey: ["admin-products"] });
   324→      navigate("/admin/products");
   325→    },
   326→    onError: (e: Error) =>
   327→      toast({ title: "Delete failed", description: e.message, variant: "destructive" }),
   328→  });
   329→
   330→  if (isLoading) return <LoadingState variant="spinner" label="Loading editor" />;
   331→  if (!allowed) return <ForbiddenState />;
   332→  if (!isNew && productQuery.isPending) return <LoadingState variant="skeleton" rows={6} />;
   333→  if (!isNew && productQuery.isError) {
   334→    return (
   335→      <AdminShell title="Edit product">
   336→        <p className="rounded-xl border border-destructive/30 bg-destructive/10 p-4 text-sm text-destructive">
   337→          This product could not be loaded.{" "}
   338→          {productQuery.error instanceof Error ? productQuery.error.message : ""}
   339→        </p>
   340→      </AdminShell>
   341→    );
   342→  }
   343→
   344→  const onSubmit = (values: ProductFormValues) => saveMutation.mutate(values);
   345→
   346→  const addGallery = () => {
   347→    const url = galleryInput.trim();
   348→    if (!url || gallery.includes(url) || gallery.length >= 8) return;
   349→    setGallery([...gallery, url]);
   350→    setGalleryInput("");
   351→    setExtraDirty(true);
   352→  };
   353→
   354→  /* ---------------- AI assist wiring ---------------- */
   355→
   356→  const buildAiRequest = (action: AiAssistAction): Record<string, unknown> => {
   357→    if (action !== "description") throw new Error("This action is not available for products.");
   358→    const name = (form.getValues("name") ?? "").trim();
   359→    if (!name) throw new Error("Add the product name first.");
   360→    return {
   361→      action,
   362→      productName: name,
   363→      tagline: (form.getValues("tagline") ?? "").trim() || undefined,
   364→      specs: Object.fromEntries(
   365→        keySpecs.filter((row) => row.key.trim()).map((row) => [row.key.trim(), row.value])
   366→      ),
   367→      pros: pros.map((s) => s.trim()).filter(Boolean),
   368→      cons: cons.map((s) => s.trim()).filter(Boolean),
   369→    };
   370→  };
   371→
   372→  const applyAiResult = (action: AiAssistAction, text: string) => {
   373→    if (action !== "description") return;
   374→    form.setValue("description", text, { shouldDirty: true });
   375→    setDescTab("write");
   376→    toast({ title: "Description drafted", description: "Review the copy before saving." });
   377→  };
   378→
   379→  const aiConfirmFor = (action: AiAssistAction): AiAssistConfirm | null => {
   380→    if (action !== "description") return null;
   381→    if (!(form.getValues("description") ?? "").trim()) return null; // nothing to lose yet
   382→    return {
   383→      title: "Replace the current description?",
   384→      description: "The markdown description will be replaced by the AI-drafted review. The previous version stays recoverable from your local draft or the server copy until you save.",
   385→    };
   386→  };
   387→
   388→  const rowEditor = (
   389→    rows: string[],
   390→    setRows: (rows: string[]) => void,
   391→    label: string,
   392→    placeholder: string
   393→  ) => (
   394→    <div className="space-y-2">
   395→      {rows.map((row, i) => (
   396→        <div key={`${label}-${i}`} className="flex gap-2">
   397→          <Input
   398→            value={row}
   399→            onChange={(e) => {
   400→              const next = [...rows];
   401→              next[i] = e.target.value;
   402→              setRows(next);
   403→              setExtraDirty(true);
   404→            }}
   405→            placeholder={placeholder}
   406→            aria-label={`${label} ${i + 1}`}
   407→            className="h-10"
   408→          />
   409→          <Button
   410→            type="button"
   411→            variant="ghost"
   412→            size="icon"
   413→            className="size-10 shrink-0 text-muted-foreground hover:text-destructive"
   414→            onClick={() => {
   415→              setRows(rows.filter((_, idx) => idx !== i));
   416→              setExtraDirty(true);
   417→            }}
   418→            aria-label={`Remove ${label} ${i + 1}`}
   419→          >
   420→            <X className="size-4" aria-hidden="true" />
   421→          </Button>
   422→        </div>
   423→      ))}
   424→      <Button
   425→        type="button"
   426→        variant="outline"
   427→        size="sm"
   428→        className="h-9 gap-2"
   429→        onClick={() => {
   430→          setRows([...rows, ""]);
   431→          setExtraDirty(true);
   432→        }}
   433→      >
   434→        <Plus className="size-4" aria-hidden="true" />
   435→        Add {label.toLowerCase()}
   436→      </Button>
   437→    </div>
   438→  );
   439→
   440→  return (
   441→    <AdminShell
   442→      title={isNew ? "New product" : "Edit product"}
   443→      description={isNew ? "Add a curated item to the affiliate store." : `Last updated ${fmtDate(productQuery.data?.updatedAt)}`}
   444→      actions={
   445→        <>
   446→          {isDirty ? (
   447→            <Badge variant="outline" className="h-9 border-amber-500/40 bg-amber-500/10 px-3 text-amber-600 dark:text-amber-400">
   448→              Unsaved changes
   449→            </Badge>
   450→          ) : null}
   451→          <ALink href="#/admin/products">
   452→            <Button variant="outline" size="sm" className="h-9 gap-2">
   453→              <ArrowLeft className="size-4" aria-hidden="true" />
   454→              All products
   455→            </Button>
   456→          </ALink>
   457→          <Button
   458→            size="sm"
   459→            className="h-9"
   460→            onClick={form.handleSubmit(onSubmit)}
   461→            disabled={saveMutation.isPending}
   462→          >
   463→            {saveMutation.isPending ? "Saving..." : isNew ? "Create product" : "Save changes"}
   464→          </Button>
   465→          {!isNew ? (
   466→            <ConfirmAction
   467→              trigger={
   468→                <Button
   469→                  variant="outline"
   470→                  size="icon"
   471→                  className="size-9 text-muted-foreground hover:text-destructive"
   472→                  aria-label="Delete product"
   473→                >
   474→                  <Trash2 className="size-4" aria-hidden="true" />
   475→                </Button>
   476→              }
   477→              title="Delete this product?"
   478→              description="The catalog entry and its click history will be permanently removed."
   479→              onConfirm={() => deleteMutation.mutateAsync()}
   480→            />
   481→          ) : null}
   482→        </>
   483→      }
   484→    >
   485→      <SEOHead title={`${isNew ? "New product" : "Edit product"} — Admin & Developer | MN.KP`} noindex />
   486→
   487→      {draft.snapshot ? (
   488→        <DraftRecoveryBanner
   489→          savedAt={draft.snapshot.savedAt}
   490→          onRestore={draft.restore}
   491→          onDiscard={draft.discard}
   492→        />
   493→      ) : null}
   494→
   495→      <Form {...form}>
   496→        <form onSubmit={form.handleSubmit(onSubmit)} className="grid gap-6 lg:grid-cols-[1fr_360px]">
   497→          {/* ---------------- main column ---------------- */}
   498→          <div className="space-y-6">
   499→            <Card>
   500→              <CardContent className="space-y-5 pt-6">
   501→                <FormField
   502→                  control={form.control}
   503→                  name="name"
   504→                  render={({ field }) => (
   505→                    <FormItem>
   506→                      <FormLabel>Name</FormLabel>
   507→                      <FormControl>
   508→                        <Input placeholder="Sony WH-1000XM5 Wireless Headphones" {...field} className="h-10 text-base" />
   509→                      </FormControl>
   510→                      <FormMessage />
   511→                    </FormItem>
   512→                  )}
   513→                />
   514→
   515→                <FormField
   516→                  control={form.control}
   517→                  name="slug"
   518→                  render={({ field }) => (
   519→                    <FormItem>
   520→                      <FormLabel>Slug</FormLabel>
   521→                      <FormControl>
   522→                        <div className="flex items-center gap-2">
   523→                          <span className="hidden shrink-0 text-xs text-muted-foreground sm:inline">/store/</span>
   524→                          <Input
   525→                            {...field}
   526→                            value={slugTouched ? field.value : effectiveSlug}
   527→                            onChange={(e) => {
   528→                              setSlugTouched(true);
   529→                              field.onChange(e);
   530→                            }}
   531→                            placeholder="auto-generated-from-name"
   532→                            className="h-10 font-mono text-sm"
   533→                          />
   534→                        </div>
   535→                      </FormControl>
   536→                      <FormDescription>
   537→                        {slugTouched ? "Custom slug — dashes and lowercase" : "Auto-generated from the name (editable)"}
   538→                      </FormDescription>
   539→                      <FormMessage />
   540→                    </FormItem>
   541→                  )}
   542→                />
   543→
   544→                <FormField
   545→                  control={form.control}
   546→                  name="tagline"
   547→                  render={({ field }) => (
   548→                    <FormItem>
