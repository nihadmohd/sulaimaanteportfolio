    49→    49→-- Auto-create a profile whenever Supabase Auth creates a user
    50→    50→create or replace function public.handle_new_user()
    51→    51→returns trigger
    52→    52→language plpgsql security definer set search_path = public
    53→    53→as $$
    54→    54→begin
    55→    55→  insert into public.profiles (id, email, full_name)
    56→    56→  values (
    57→    57→    new.id,
    58→    58→    new.email,
    59→    59→    coalesce(new.raw_user_meta_data ->> 'full_name', split_part(new.email, '@', 1))
    60→    60→  )
    61→    61→  on conflict (id) do nothing;
    62→    62→  return new;
    63→    63→end;
    64→    64→$$;

    83→    83→-- role gate used by RLS policies (security definer avoids RLS recursion)
    84→    84→create or replace function public.is_admin()
    85→    85→returns boolean
    86→    86→language sql stable security definer set search_path = public
    87→    87→as $$
    88→    88→  select coalesce(
    89→    89→    exists (
    90→    90→      select 1 from public.profiles p
    91→    91→      where p.id = auth.uid() and p.role in ('editor','admin')
    92→    92→    ), false
    93→    93→  );
    94→    94→$$;

    96→    96→-- block role / activation self-escalation on profiles
    97→    97→create or replace function public.guard_profile_mutations()
    98→    98→returns trigger
    99→    99→language plpgsql security definer set search_path = public
   100→   100→as $$
   101→   101→begin
   102→   102→  if (new.role <> old.role or new.is_active <> old.is_active)
   103→   103→     and not public.is_admin() then
   104→   104→    raise exception 'SECURITY: role/is_active may only be changed by an admin';
   105→   105→  end if;
   106→   106→  return new;
   107→   107→end;
   108→   108→$$;

   362→   362→  select count(*)::int as online
   363→   363→  from public.visitor_sessions
   364→   364→  where last_seen_at > now() - interval '75 seconds';
   365→   365→
   366→   366→-- callable by anon/authenticated (REST fallback for the realtime counter)
   367→   367→create or replace function public.get_online_count()
   368→   368→returns integer
   369→   369→language sql stable security definer set search_path = public
   370→   370→as $$
   371→   371→  select count(*)::int
   372→   372→  from public.visitor_sessions
   373→   373→  where last_seen_at > now() - interval '75 seconds';
   374→   374→$$;
   375→   375→
   376→   376→-- offline sweep — schedule via pg_cron, or call from an edge function
   377→   377→create or replace function public.sweep_presence()
   378→   378→returns integer language plpgsql as $$
   379→   379→declare n integer;
   380→   380→begin
   381→   381→  update public.visitor_sessions
   382→   382→     set is_online = false
   383→   383→   where is_online and last_seen_at < now() - interval '75 seconds';
   384→   384→  get diagnostics n = row_count;
   385→   385→  return n;
   386→   386→end;
   387→   387→$$;
