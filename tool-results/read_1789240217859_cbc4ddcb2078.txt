    82→const postFormSchema = z.object({
    83→  title: z.string().min(4, "Title must be at least 4 characters").max(140, "Keep the title under 140 characters"),
    84→  slug: z.string().max(96).optional(),
    85→  excerpt: z.string().max(300, "Excerpts max out at 300 characters").optional(),
    86→  content: z.string().min(1, "Content is required"),
    87→  coverImageUrl: z.string().optional(),
    88→  status: z.enum(["draft", "published", "archived"]),
    89→  isFeatured: z.boolean(),
    90→  categoryId: z.string().optional(),
    91→  publishedAt: z.string().optional(),
    92→  seoTitle: z.string().max(70, "SEO titles max out at 70 characters").optional(),
    93→  seoDescription: z.string().max(180, "SEO descriptions max out at 180 characters").optional(),
    94→  ogImageUrl: z.string().optional(),
    95→  canonicalUrl: z.string().optional(),
    96→});
    97→
    98→type PostFormValues = z.infer<typeof postFormSchema>;
    99→
   100→function wordsPerMinute(content: string): number {
   101→  const words = content.trim().split(/\s+/).filter(Boolean).length;
   102→  return Math.max(1, Math.ceil(words / 200));
   103→}
   104→
   105→export default function PostEditView() {
   106→  const { id } = useHashParams<{ id: string }>();
   107→  const isNew = !id || id === "new";
   108→  const { isLoading, allowed } = useAdminGuard();
   109→  const queryClient = useQueryClient();
   110→  const categories = useCategories("blog");
   111→
   112→  const [tags, setTags] = React.useState<string[]>([]);
   113→  const [slugTouched, setSlugTouched] = React.useState(false);
   114→  const [tagInput, setTagInput] = React.useState("");
   115→  const [contentTab, setContentTab] = React.useState<"write" | "preview">("write");
   116→  const [seoOpen, setSeoOpen] = React.useState(false);
   117→  const [extraDirty, setExtraDirty] = React.useState(false);
   118→
   119→  const form = useForm<PostFormValues>({
   120→    resolver: zodResolver(postFormSchema),
   121→    defaultValues: {
   122→      title: "",
   123→      slug: "",
   124→      excerpt: "",
   125→      content: "",
   126→      coverImageUrl: "",
   127→      status: "draft",
   128→      isFeatured: false,
   129→      categoryId: "",
   130→      publishedAt: "",
   131→      seoTitle: "",
   132→      seoDescription: "",
   133→      ogImageUrl: "",
   134→      canonicalUrl: "",
   135→    },
   136→  });
   137→
   138→  const postQuery = useQuery({
   139→    queryKey: ["admin-post", id],
   140→    queryFn: () => apiFetch<PostDTO>(`/api/posts/${id}`),
   141→    enabled: allowed && !isNew,
   142→    retry: 1,
   143→  });
   144→
   145→  React.useEffect(() => {
   146→    if (!postQuery.data) return;
   147→    const p = postQuery.data;
   148→    form.reset({
   149→      title: p.title,
   150→      slug: p.slug,
   151→      excerpt: p.excerpt ?? "",
   152→      content: p.content,
   153→      coverImageUrl: p.coverImageUrl ?? "",
   154→      status: p.status,
   155→      isFeatured: p.isFeatured,
   156→      categoryId: p.category?.id ?? "",
   157→      publishedAt: p.publishedAt ? p.publishedAt.slice(0, 10) : "",
   158→      seoTitle: p.seoTitle ?? "",
   159→      seoDescription: p.seoDescription ?? "",
   160→      ogImageUrl: p.ogImageUrl ?? "",
   161→      canonicalUrl: p.canonicalUrl ?? "",
   162→    });
   163→    setTags(p.tags);
   164→    setSlugTouched(true);
   165→    setExtraDirty(false);
   166→  }, [postQuery.data, form]);
   167→
   168→  // useWatch (single call, whole values) instead of form.watch — satisfies
   169→  // react-hooks/incompatible-library; defaults are complete so the cast is safe.
   170→  const watched = useWatch({ control: form.control }) as PostFormValues;
   171→
   172→  const effectiveSlug = slugTouched && watched.slug ? slugify(watched.slug) : slugify(watched.title);
   173→  const readingMinutes = wordsPerMinute(watched.content ?? "");
   174→  const isDirty = form.formState.isDirty || extraDirty;
   175→
   176→  const saveMutation = useMutation({
   177→    mutationFn: (values: PostFormValues) => {
   178→      const body: Record<string, unknown> = {
   179→        title: values.title,
   180→        content: values.content,
   181→        status: values.status,
   182→        isFeatured: values.isFeatured,
   183→        tags,
   184→        categoryId: values.categoryId || null,
   185→        coverImageUrl: values.coverImageUrl || "",
   186→        excerpt: values.excerpt || undefined,
   187→        publishedAt: values.publishedAt ? `${values.publishedAt}T12:00:00` : undefined,
   188→        seoTitle: values.seoTitle || undefined,
   189→        seoDescription: values.seoDescription || undefined,
   190→        ogImageUrl: values.ogImageUrl || "",
   191→        canonicalUrl: values.canonicalUrl || "",
   192→      };
   193→      if (slugTouched && values.slug) body.slug = slugify(values.slug);
   194→      return isNew
   195→        ? apiFetch<PostDTO>("/api/posts", { method: "POST", body: JSON.stringify(body) })
   196→        : apiFetch<PostDTO>(`/api/posts/${id}`, { method: "PATCH", body: JSON.stringify(body) });
   197→    },
   198→    onSuccess: (post) => {
   199→      toast({
   200→        title: isNew ? "Post created" : "Post saved",
   201→        description: post.title,
   202→      });
   203→      setExtraDirty(false);
   204→      void queryClient.invalidateQueries({ queryKey: ["admin-posts"] });
   205→      void queryClient.invalidateQueries({ queryKey: ["admin-stats"] });
   206→      if (isNew) navigate(`/admin/posts/${post.id}`);
   207→      else form.reset(form.getValues());
   208→    },
   209→    onError: (e: Error) =>
   210→      toast({ title: "Save failed", description: e.message, variant: "destructive" }),
   211→  });
   212→
   213→  const deleteMutation = useMutation({
   214→    mutationFn: () => apiFetch(`/api/posts/${id}`, { method: "DELETE" }),
   215→    onSuccess: () => {
   216→      toast({ title: "Post deleted" });
   217→      void queryClient.invalidateQueries({ queryKey: ["admin-posts"] });
   218→      navigate("/admin/posts");
   219→    },
   220→    onError: (e: Error) =>
   221→      toast({ title: "Delete failed", description: e.message, variant: "destructive" }),
   222→  });

   224→  const addTag = () => {

   228→    setTagInput("");

   230→  };

   232→  if (isLoading) return <LoadingState variant="spinner" label="Loading editor" />;
   233→  if (!allowed) return <ForbiddenState />;

   235→  if (!isNew && postQuery.isError) {
   236→    return (
   237→      <AdminShell title="Edit post">
   238→        <p className="rounded-xl border border-destructive/30 bg-destructive/10 p-4 text-sm text-destructive">

   241→      </AdminShell>

   243→  }

   245→  const onSubmit = (values: PostFormValues) => saveMutation.mutate(values);

   248→  const serpDesc = (watched.seoDescription || watched.excerpt || "Add an excerpt or SEO description...").slice(0, 155);

   252→      title={isNew ? "New post" : "Edit post"}

   257→            <Badge variant="outline" className="h-9 border-amber-500/40 bg-amber-500/10 px-3 text-amber-600 dark:text-amber-400">

   259→            </Badge>

   263→              <ArrowLeft className="size-4" aria-hidden="true" />
   264→              All posts
   265→            </Button>
   266→          </ALink>
   267→          <Button
   268→            size="sm"
   269→            className="h-9"
   270→            onClick={form.handleSubmit(onSubmit)}
   271→            disabled={saveMutation.isPending}
   272→          >
   273→            {saveMutation.isPending ? "Saving..." : isNew ? "Create post" : "Save changes"}
   274→          </Button>
   275→          {!isNew ? (
   276→            <ConfirmAction
   277→              trigger={
   278→                <Button
   279→                  variant="outline"
   280→                  size="icon"
   281→                  className="size-9 text-muted-foreground hover:text-destructive"
   282→                  aria-label="Delete post"
   283→                >
   284→                  <Trash2 className="size-4" aria-hidden="true" />
   285→                </Button>
   286→              }
   287→              title="Delete this post?"
   288→              description="The post and its view history will be permanently removed."
   289→              onConfirm={() => deleteMutation.mutateAsync()}
   290→            />
   291→          ) : null}
   292→        </>
   293→      }
   294→    >
   295→      <SEOHead title={`${isNew ? "New post" : "Edit post"} — Admin & Developer | MN.KP`} noindex />
   296→
   297→      <Form {...form}>
   298→        <form onSubmit={form.handleSubmit(onSubmit)} className="grid gap-6 lg:grid-cols-[1fr_380px]">
   299→          {/* ---------------- main form column ---------------- */}
   300→          <div className="space-y-6">
   301→            <Card>
   302→              <CardContent className="space-y-5 pt-6">
   303→                <FormField
   304→                  control={form.control}
   305→                  name="title"
   306→                  render={({ field }) => (
   307→                    <FormItem>
   308→                      <FormLabel>Title</FormLabel>
   309→                      <FormControl>
   310→                        <Input placeholder="An honest headline that earns the click" {...field} className="h-10 text-base" />
   311→                      </FormControl>
   312→                      <FormDescription>
   313→                        {field.value?.length ?? 0}/140 characters · becomes the H1 and SERP fallback
   314→                      </FormDescription>
   315→                      <FormMessage />
   316→                    </FormItem>
   317→                  )}
   318→                />
   319→
   320→                <FormField
   321→                  control={form.control}
   322→                  name="slug"
   323→                  render={({ field }) => (
   324→                    <FormItem>
   325→                      <FormLabel>Slug</FormLabel>
   326→                      <FormControl>
   327→                        <div className="flex items-center gap-2">
   328→                          <span className="hidden shrink-0 text-xs text-muted-foreground sm:inline">/blog/</span>
   329→                          <Input
   330→                            {...field}
   331→                            value={slugTouched ? field.value : effectiveSlug}
   332→                            onChange={(e) => {
   333→                              setSlugTouched(true);
   334→                              field.onChange(e);
   335→                            }}
   336→                            placeholder="auto-generated-from-title"
   337→                            className="h-10 font-mono text-sm"
   338→                          />
   339→                        </div>
   340→                      </FormControl>
   341→                      <FormDescription>
   342→                        {slugTouched ? "Custom slug — dashes and lowercase" : "Auto-generated from the title (editable)"}
   343→                      </FormDescription>
   344→                      <FormMessage />
   345→                    </FormItem>
   346→                  )}
   347→                />
   348→
   349→                <FormField
   350→                  control={form.control}
   351→                  name="excerpt"
   352→                  render={({ field }) => (
   353→                    <FormItem>
   354→                      <FormLabel>Excerpt</FormLabel>
   355→                      <FormControl>
   356→                        <Textarea
   357→                          {...field}
   358→                          value={field.value ?? ""}
   359→                          rows={2}
   360→                          placeholder="One or two sentences for cards and search results"
   361→                        />
   362→                      </FormControl>
   363→                      <FormDescription>{(field.value ?? "").length}/300 characters</FormDescription>
   364→                      <FormMessage />
   365→                    </FormItem>
   366→                  )}
   367→                />
   368→
   369→                <div>
   370→                  <FormLabel>Content (markdown)</FormLabel>
   371→                  <Tabs
   372→                    value={contentTab}
   373→                    onValueChange={(v) => setContentTab(v as "write" | "preview")}
   374→                    className="mt-2"
   375→                  >
   376→                    <div className="flex items-center justify-between">
   377→                      <TabsList className="h-9">
   378→                        <TabsTrigger value="write" className="h-7">Write</TabsTrigger>
   379→                        <TabsTrigger value="preview" className="h-7">
   380→                          <Eye className="mr-1 size-3.5" aria-hidden="true" />
   381→                          Preview
   382→                        </TabsTrigger>
   383→                      </TabsList>
   384→                      <p className="text-xs text-muted-foreground">
   385→                        ~{readingMinutes} min read · {(watched.content ?? "").length.toLocaleString("en-IN")} chars
   386→                      </p>
   387→                    </div>
   388→                  </Tabs>
   389→                  {contentTab === "write" ? (
   390→                    <FormField
   391→                      control={form.control}
   392→                      name="content"
   393→                      render={({ field }) => (
   394→                        <FormItem className="mt-3">
   395→                          <FormControl>
   396→                            <Textarea
   397→                              {...field}
   398→                              rows={18}
   399→                              placeholder={"## A question-style heading works great for AI answers\n\nWrite the article in markdown..."}
   400→                              className="min-h-[420px] font-mono text-[13px] leading-relaxed"
   401→                            />
   402→                          </FormControl>
   403→                          <FormMessage />
   404→                        </FormItem>
   405→                      )}
   406→                    />
   407→                  ) : (
   408→                    <div className="mt-3 min-h-[420px] rounded-lg border p-4">
   409→                      {(watched.content ?? "").trim() ? (
   410→                        <React.Suspense fallback={<MarkdownFallback />}>
   411→                          <MarkdownBlock content={watched.content} />
   412→                        </React.Suspense>
   413→                      ) : (
   414→                        <p className="text-sm text-muted-foreground">Nothing to preview yet.</p>
   415→                      )}
   416→                    </div>
   417→                  )}
   418→                </div>
   419→
   420→                <FormField
   421→                  control={form.control}
   422→                  name="coverImageUrl"
   423→                  render={({ field }) => (
   424→                    <FormItem>
   425→                      <FormLabel>Cover image URL</FormLabel>
   426→                      <FormControl>
   427→                        <Input {...field} value={field.value ?? ""} placeholder="/images/blog/... or https://..." className="h-10" />
   428→                      </FormControl>
   429→                      <FormDescription>Quick pick:</FormDescription>
   430→                      <div className="flex flex-wrap gap-2 pt-1">
   431→                        {BLOG_COVERS.map((src) => (
   432→                          <button
   433→                            key={src}
   434→                            type="button"
   435→                            onClick={() => {
   436→                              field.onChange(src);
   437→                              setExtraDirty(true);
   438→                            }}
   439→                            aria-label={`Use cover ${src}`}
   440→                            className={`size-14 overflow-hidden rounded-md border-2 transition-transform hover:scale-105 ${
   441→                              field.value === src ? "border-gold" : "border-transparent"
   442→                            }`}
   443→                          >
   444→                            <img src={src} alt="" loading="lazy" decoding="async" className="size-full object-cover" />
   445→                          </button>
   446→                        ))}
   447→                      </div>
   448→                      <FormMessage />
   449→                    </FormItem>
   450→                  )}
   451→                />
   452→
   453→                <div>
   454→                  <FormLabel>Tags</FormLabel>
   455→                  <div className="mt-2 flex gap-2">
   456→                    <Input
   457→                      value={tagInput}
   458→                      onChange={(e) => setTagInput(e.target.value)}
   459→                      onKeyDown={(e) => {
   460→                        if (e.key === "Enter" || e.key === ",") {
   461→                          e.preventDefault();
   462→                          addTag();
   463→                        }
   464→                      }}
   465→                      placeholder="Type a tag, press Enter (max 10)"
   466→                      aria-label="Add tag"
   467→                      className="h-10"
   468→                    />
   469→                    <Button type="button" variant="outline" className="h-10" onClick={addTag}>
   470→                      Add
   471→                    </Button>
   472→                  </div>
   473→                  {tags.length > 0 ? (
   474→                    <div className="mt-3 flex flex-wrap gap-2">
   475→                      {tags.map((tag) => (
   476→                        <Badge key={tag} variant="secondary" className="gap-1 pr-1.5">
   477→                          {tag}
   478→                          <button
   479→                            type="button"
   480→                            onClick={() => {
   481→                              setTags(tags.filter((t) => t !== tag));
   482→                              setExtraDirty(true);
   483→                            }}
   484→                            aria-label={`Remove tag ${tag}`}
   485→                            className="flex size-5 items-center justify-center rounded-full hover:bg-muted"
   486→                          >
   487→                            <X className="size-3" aria-hidden="true" />
   488→                          </button>
   489→                        </Badge>
   490→                      ))}
   491→                    </div>
   492→                  ) : (
   493→                    <p className="mt-2 text-xs text-muted-foreground">No tags yet — tags power related posts and filters.</p>
   494→                  )}
   495→                </div>
   496→              </CardContent>
   497→            </Card>
   498→
   499→            {/* ---------------- SEO card ---------------- */}
   500→            <Collapsible open={seoOpen} onOpenChange={setSeoOpen}>
   501→              <Card className="border-gold/40">
   502→                <CardHeader className="pb-3">
   503→                  <CollapsibleTrigger asChild>
   504→                    <button
   505→                      type="button"
   506→                      className="flex w-full items-center justify-between text-left"
   507→                      aria-expanded={seoOpen}
   508→                    >
   509→                      <div>
   510→                        <CardTitle className="text-base text-gold">SEO &amp; sharing</CardTitle>
   511→                        <p className="mt-0.5 text-xs text-muted-foreground">
   512→                          Search title, description, OG image and canonical URL
   513→                        </p>
   514→                      </div>
   515→                      <ChevronDown
   516→                        className={`size-5 text-gold transition-transform ${seoOpen ? "rotate-180" : ""}`}
   517→                        aria-hidden="true"
   518→                      />
   519→                    </button>
   520→                  </CollapsibleTrigger>
   521→                </CardHeader>
   522→                <CollapsibleContent>
   523→                  <CardContent className="space-y-5 pt-2">
   524→                    <FormField
   525→                      control={form.control}
   526→                      name="seoTitle"
   527→                      render={({ field }) => (
   528→                        <FormItem>
   529→                          <FormLabel>SEO title</FormLabel>
   530→                          <FormControl>
   531→                            <Input {...field} value={field.value ?? ""} placeholder="Falls back to the post title" className="h-10" />
   532→                          </FormControl>
   533→                          <FormDescription>{(field.value ?? "").length}/60 recommended</FormDescription>
   534→                          <FormMessage />
   535→                        </FormItem>
   536→                      )}
   537→                    />
   538→                    <FormField
   539→                      control={form.control}
   540→                      name="seoDescription"
   541→                      render={({ field }) => (
   542→                        <FormItem>
   543→                          <FormLabel>SEO description</FormLabel>
   544→                          <FormControl>
   545→                            <Textarea
   546→                              {...field}
   547→                              value={field.value ?? ""}
   548→                              rows={3}
   549→                              placeholder="Falls back to the excerpt"
   550→                            />
   551→                          </FormControl>
   552→                          <FormDescription>{(field.value ?? "").length}/155 recommended</FormDescription>
   553→                          <FormMessage />

   556→                    />
   557→                    <FormField

   560→                      render={({ field }) => (
   561→                        <FormItem>
   562→                          <FormLabel>OG / social image URL</FormLabel>
   563→                          <FormControl>
