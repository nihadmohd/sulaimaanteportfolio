    76→
    77→const MERCHANT_SUGGESTIONS = ["Amazon", "Flipkart", "MN.KP Digital", "Croma", "Myntra", "Noise"];
    78→
    79→const productFormSchema = z.object({
    80→  name: z.string().min(3, "Product name is required").max(140),
    81→  slug: z.string().max(96).optional(),
    82→  tagline: z.string().max(160).optional(),
    83→  description: z.string().optional(),
    84→  brand: z.string().max(60).optional(),
    85→  merchant: z.string().max(60).optional(),
    86→  imageUrl: z.string().optional(),
    87→  affiliateUrl: z.string().min(3, "Affiliate URL is required").max(600),
    88→  price: z
    89→    .string()
    90→    .optional()
    91→    .refine((v) => !v || (!Number.isNaN(Number(v)) && Number(v) >= 0), "Enter a valid price"),
    92→  compareAtPrice: z
    93→    .string()
    94→    .optional()
    95→    .refine((v) => !v || (!Number.isNaN(Number(v)) && Number(v) >= 0), "Enter a valid price"),
    96→  reviewCount: z
    97→    .string()
    98→    .optional()
    99→    .refine((v) => !v || (Number.isInteger(Number(v)) && Number(v) >= 0), "Whole number"),
   100→  status: z.enum(["active", "draft", "archived"]),
   101→  isFeatured: z.boolean(),
   102→  categoryId: z.string().optional(),
   103→});
   104→
   105→type ProductFormValues = z.infer<typeof productFormSchema>;
   106→
   107→interface SpecRow {
   108→  key: string;
   109→  value: string;
   110→}
   111→
   112→export default function ProductEditView() {
   113→  const { id } = useHashParams<{ id: string }>();
   114→  const isNew = !id || id === "new";
   115→  const { isLoading, allowed } = useAdminGuard();
   116→  const queryClient = useQueryClient();
   117→  const categories = useCategories("store");
   118→
   119→  const [gallery, setGallery] = React.useState<string[]>([]);
   120→  const [pros, setPros] = React.useState<string[]>([]);
   121→  const [cons, setCons] = React.useState<string[]>([]);
   122→  const [keySpecs, setKeySpecs] = React.useState<SpecRow[]>([]);
   123→  const [rating, setRating] = React.useState(0);
   124→  const [slugTouched, setSlugTouched] = React.useState(false);
   125→  const [galleryInput, setGalleryInput] = React.useState("");
   126→  const [descTab, setDescTab] = React.useState<"write" | "preview">("write");
   127→  const [extraDirty, setExtraDirty] = React.useState(false);
   128→
   129→  const form = useForm<ProductFormValues>({
   130→    resolver: zodResolver(productFormSchema),
   131→    defaultValues: {
   132→      name: "",
   133→      slug: "",
   134→      tagline: "",
   135→      description: "",
   136→      brand: "",
   137→      merchant: "",
   138→      imageUrl: "",
   139→      affiliateUrl: "",
   140→      price: "",
   141→      compareAtPrice: "",
   142→      reviewCount: "",
   143→      status: "active",
   144→      isFeatured: false,
   145→      categoryId: "",
   146→    },
   147→  });
   148→
   149→  const productQuery = useQuery({
   150→    queryKey: ["admin-product", id],
   151→    queryFn: () => apiFetch<ProductDTO>(`/api/products/${id}`),
   152→    enabled: allowed && !isNew,
   153→    retry: 1,
   154→  });
   155→
   156→  React.useEffect(() => {
   157→    if (!productQuery.data) return;
   158→    const p = productQuery.data;
   159→    form.reset({
   160→      name: p.name,
   161→      slug: p.slug,
   162→      tagline: p.tagline ?? "",
   163→      description: p.description ?? "",
   164→      brand: p.brand ?? "",
   165→      merchant: p.merchant ?? "",
   166→      imageUrl: p.imageUrl ?? "",
   167→      affiliateUrl: p.affiliateUrl,
   168→      price: p.price != null ? String(p.price) : "",
   169→      compareAtPrice: p.compareAtPrice != null ? String(p.compareAtPrice) : "",
   170→      reviewCount: String(p.reviewCount ?? 0),
   171→      status: p.status,
   172→      isFeatured: p.isFeatured,
   173→      categoryId: p.category?.id ?? "",
   174→    });
   175→    setGallery(p.gallery);
   176→    setPros(p.pros);
   177→    setCons(p.cons);
   178→    setKeySpecs(Object.entries(p.keySpecs).map(([key, value]) => ({ key, value })));
   179→    setRating(p.rating);
   180→    setSlugTouched(true);
   181→    setExtraDirty(false);
   182→  }, [productQuery.data, form]);
   183→
   184→  // useWatch (single call, whole values) instead of form.watch — satisfies
   185→  // react-hooks/incompatible-library; defaults are complete so the cast is safe.
   186→  const watched = useWatch({ control: form.control }) as ProductFormValues;
   187→
   188→  const effectiveSlug = slugTouched && watched.slug ? slugify(watched.slug) : slugify(watched.name);
   189→  const priceNum = watched.price ? Number(watched.price) : 0;
   190→  const compareNum = watched.compareAtPrice ? Number(watched.compareAtPrice) : 0;
   191→  const discount =
   192→    compareNum > priceNum && priceNum > 0 ? Math.round((1 - priceNum / compareNum) * 100) : 0;
   193→  const isDirty = form.formState.isDirty || extraDirty;
   194→
   195→  const saveMutation = useMutation({
   196→    mutationFn: (values: ProductFormValues) => {
   197→      const body: Record<string, unknown> = {
   198→        name: values.name,
   199→        affiliateUrl: values.affiliateUrl,
   200→        status: values.status,
   201→        isFeatured: values.isFeatured,
   202→        currency: "INR",
   203→        imageUrl: values.imageUrl || "",
   204→        gallery: gallery.filter(Boolean),
   205→        pros: pros.map((s) => s.trim()).filter(Boolean),
   206→        cons: cons.map((s) => s.trim()).filter(Boolean),
   207→        keySpecs: Object.fromEntries(
   208→          keySpecs.filter((row) => row.key.trim()).map((row) => [row.key.trim(), row.value])
   209→        ),
   210→        rating,
   211→        reviewCount: Number(values.reviewCount || 0),
   212→        price: values.price ? Number(values.price) : null,
   213→        compareAtPrice: values.compareAtPrice ? Number(values.compareAtPrice) : null,
   214→        categoryId: values.categoryId || null,
   215→        tagline: values.tagline || undefined,
   216→        description: values.description || undefined,
   217→        brand: values.brand || undefined,
   218→        merchant: values.merchant || undefined,
   219→      };
   220→      if (slugTouched && values.slug) body.slug = slugify(values.slug);
   221→      return isNew
   222→        ? apiFetch<ProductDTO>("/api/products", { method: "POST", body: JSON.stringify(body) })
   223→        : apiFetch<ProductDTO>(`/api/products/${id}`, { method: "PATCH", body: JSON.stringify(body) });
   224→    },
   225→    onSuccess: (product) => {
   226→      toast({ title: isNew ? "Product created" : "Product saved", description: product.name });
   227→      setExtraDirty(false);
   228→      void queryClient.invalidateQueries({ queryKey: ["admin-products"] });
   229→      void queryClient.invalidateQueries({ queryKey: ["admin-stats"] });
   230→      if (isNew) navigate(`/admin/products/${product.id}`);
   231→      else form.reset(form.getValues());
   232→    },
   233→    onError: (e: Error) =>
   234→      toast({ title: "Save failed", description: e.message, variant: "destructive" }),
   235→  });
   236→
   237→  const deleteMutation = useMutation({
   238→    mutationFn: () => apiFetch(`/api/products/${id}`, { method: "DELETE" }),
   239→    onSuccess: () => {
   240→      toast({ title: "Product deleted" });
   241→      void queryClient.invalidateQueries({ queryKey: ["admin-products"] });
   242→      navigate("/admin/products");
   243→    },
   244→    onError: (e: Error) =>
   245→      toast({ title: "Delete failed", description: e.message, variant: "destructive" }),
   246→  });
   247→
   248→  if (isLoading) return <LoadingState variant="spinner" label="Loading editor" />;
   249→  if (!allowed) return <ForbiddenState />;
   250→  if (!isNew && productQuery.isPending) return <LoadingState variant="skeleton" rows={6} />;
   251→  if (!isNew && productQuery.isError) {
   252→    return (
   253→      <AdminShell title="Edit product">
   254→        <p className="rounded-xl border border-destructive/30 bg-destructive/10 p-4 text-sm text-destructive">
   255→          This product could not be loaded.{" "}
   256→          {productQuery.error instanceof Error ? productQuery.error.message : ""}
   257→        </p>
   258→      </AdminShell>
   259→    );
   260→  }
   261→
   262→  const onSubmit = (values: ProductFormValues) => saveMutation.mutate(values);
   263→
   264→  const addGallery = () => {
   265→    const url = galleryInput.trim();
   266→    if (!url || gallery.includes(url) || gallery.length >= 8) return;
   267→    setGallery([...gallery, url]);
   268→    setGalleryInput("");
   269→    setExtraDirty(true);
   270→  };
   271→
   272→  const rowEditor = (
   273→    rows: string[],
   274→    setRows: (rows: string[]) => void,
   275→    label: string,
   276→    placeholder: string
   277→  ) => (
   278→    <div className="space-y-2">
   279→      {rows.map((row, i) => (
   280→        <div key={`${label}-${i}`} className="flex gap-2">
   281→          <Input
   282→            value={row}
   283→            onChange={(e) => {
   284→              const next = [...rows];
   285→              next[i] = e.target.value;
   286→              setRows(next);
   287→              setExtraDirty(true);
   288→            }}
   289→            placeholder={placeholder}
   290→            aria-label={`${label} ${i + 1}`}
   291→            className="h-10"
   292→          />
   293→          <Button
   294→            type="button"
   295→            variant="ghost"
   296→            size="icon"
   297→            className="size-10 shrink-0 text-muted-foreground hover:text-destructive"
   298→            onClick={() => {
   299→              setRows(rows.filter((_, idx) => idx !== i));
   300→              setExtraDirty(true);
   301→            }}
   302→            aria-label={`Remove ${label} ${i + 1}`}
   303→          >
   304→            <X className="size-4" aria-hidden="true" />
   305→          </Button>
   306→        </div>
   307→      ))}
   308→      <Button
   309→        type="button"
   310→        variant="outline"
   311→        size="sm"
   312→        className="h-9 gap-2"
   313→        onClick={() => {
   314→          setRows([...rows, ""]);
   315→          setExtraDirty(true);
   316→        }}
   317→      >
   318→        <Plus className="size-4" aria-hidden="true" />
   319→        Add {label.toLowerCase()}
   320→      </Button>
   321→    </div>
   322→  );
   323→
   324→  return (
   325→    <AdminShell
   326→      title={isNew ? "New product" : "Edit product"}
   327→      description={isNew ? "Add a curated item to the affiliate store." : `Last updated ${fmtDate(productQuery.data?.updatedAt)}`}
   328→      actions={
   329→        <>
   330→          {isDirty ? (
   331→            <Badge variant="outline" className="h-9 border-amber-500/40 bg-amber-500/10 px-3 text-amber-600 dark:text-amber-400">
   332→              Unsaved changes
   333→            </Badge>
   334→          ) : null}
   335→          <ALink href="#/admin/products">
   336→            <Button variant="outline" size="sm" className="h-9 gap-2">
   337→              <ArrowLeft className="size-4" aria-hidden="true" />
   338→              All products
   339→            </Button>
   340→          </ALink>
   341→          <Button
   342→            size="sm"
   343→            className="h-9"
   344→            onClick={form.handleSubmit(onSubmit)}
   345→            disabled={saveMutation.isPending}
   346→          >
   347→            {saveMutation.isPending ? "Saving..." : isNew ? "Create product" : "Save changes"}
   348→          </Button>
   349→          {!isNew ? (
   350→            <ConfirmAction
   351→              trigger={
   352→                <Button
   353→                  variant="outline"
   354→                  size="icon"
   355→                  className="size-9 text-muted-foreground hover:text-destructive"
   356→                  aria-label="Delete product"
   357→                >
   358→                  <Trash2 className="size-4" aria-hidden="true" />
   359→                </Button>
   360→              }
   361→              title="Delete this product?"
   362→              description="The catalog entry and its click history will be permanently removed."
   363→              onConfirm={() => deleteMutation.mutateAsync()}
   364→            />
   365→          ) : null}
   366→        </>
   367→      }
   368→    >
   369→      <SEOHead title={`${isNew ? "New product" : "Edit product"} — Admin & Developer | MN.KP`} noindex />
   370→
   371→      <Form {...form}>
   372→        <form onSubmit={form.handleSubmit(onSubmit)} className="grid gap-6 lg:grid-cols-[1fr_360px]">
   373→          {/* ---------------- main column ---------------- */}
   374→          <div className="space-y-6">
   375→            <Card>
   376→              <CardContent className="space-y-5 pt-6">
   377→                <FormField
   378→                  control={form.control}
   379→                  name="name"
   380→                  render={({ field }) => (
   381→                    <FormItem>
   382→                      <FormLabel>Name</FormLabel>
   383→                      <FormControl>
   384→                        <Input placeholder="Sony WH-1000XM5 Wireless Headphones" {...field} className="h-10 text-base" />
   385→                      </FormControl>
   386→                      <FormMessage />
   387→                    </FormItem>
   388→                  )}
   389→                />
   390→
   391→                <FormField
   392→                  control={form.control}
   393→                  name="slug"
   394→                  render={({ field }) => (
   395→                    <FormItem>
   396→                      <FormLabel>Slug</FormLabel>
   397→                      <FormControl>
   398→                        <div className="flex items-center gap-2">
   399→                          <span className="hidden shrink-0 text-xs text-muted-foreground sm:inline">/store/</span>
   400→                          <Input
   401→                            {...field}
   402→                            value={slugTouched ? field.value : effectiveSlug}
   403→                            onChange={(e) => {
   404→                              setSlugTouched(true);
   405→                              field.onChange(e);
   406→                            }}
   407→                            placeholder="auto-generated-from-name"
   408→                            className="h-10 font-mono text-sm"
   409→                          />
   410→                        </div>
   411→                      </FormControl>
   412→                      <FormDescription>
   413→                        {slugTouched ? "Custom slug — dashes and lowercase" : "Auto-generated from the name (editable)"}
   414→                      </FormDescription>
   415→                      <FormMessage />
   416→                    </FormItem>
   417→                  )}
   418→                />
   419→
   420→                <FormField
   421→                  control={form.control}
   422→                  name="tagline"
   423→                  render={({ field }) => (
   424→                    <FormItem>
   425→                      <FormLabel>Tagline</FormLabel>
   426→                      <FormControl>
   427→                        <Input {...field} value={field.value ?? ""} placeholder="The one-line pitch shown on cards" className="h-10" />
   428→                      </FormControl>
   429→                      <FormDescription>{(field.value ?? "").length}/160 characters</FormDescription>
   430→                      <FormMessage />
   431→                    </FormItem>
   432→                  )}
   433→                />
   434→
   435→                <div className="grid gap-5 sm:grid-cols-2">
   436→                  <FormField
   437→                    control={form.control}
   438→                    name="brand"
   439→                    render={({ field }) => (
   440→                      <FormItem>
   441→                        <FormLabel>Brand</FormLabel>
   442→                        <FormControl>
   443→                          <Input {...field} value={field.value ?? ""} placeholder="Sony" className="h-10" />
   444→                        </FormControl>
   445→                        <FormMessage />
   446→                      </FormItem>
   447→                    )}
   448→                  />
   449→                  <FormField
   450→                    control={form.control}
   451→                    name="merchant"
   452→                    render={({ field }) => (
   453→                      <FormItem>
   454→                        <FormLabel>Merchant</FormLabel>
   455→                        <FormControl>
   456→                          <>
   457→                            <Input {...field} value={field.value ?? ""} placeholder="Amazon" list="merchant-suggestions" className="h-10" />
   458→                            <datalist id="merchant-suggestions">
   459→                              {MERCHANT_SUGGESTIONS.map((m) => (
   460→                                <option key={m} value={m} />
   461→                              ))}
   462→                            </datalist>
   463→                          </>
   464→                        </FormControl>
   465→                        <FormDescription>Suggestions: Amazon, Flipkart, MN.KP Digital...</FormDescription>
   466→                        <FormMessage />
   467→                      </FormItem>
   468→                    )}
   469→                  />
   470→                </div>
   471→
   472→                <div>
   473→                  <FormLabel>Description (markdown)</FormLabel>
   474→                  <Tabs value={descTab} onValueChange={(v) => setDescTab(v as "write" | "preview")} className="mt-2">
   475→                    <div className="flex items-center justify-between">
   476→                      <TabsList className="h-9">
   477→                        <TabsTrigger value="write" className="h-7">Write</TabsTrigger>
   478→                        <TabsTrigger value="preview" className="h-7">
   479→                          <Eye className="mr-1 size-3.5" aria-hidden="true" />
   480→                          Preview
   481→                        </TabsTrigger>
   482→                      </TabsList>
   483→                      <p className="text-xs text-muted-foreground">
   484→                        {(watched.description ?? "").length.toLocaleString("en-IN")} chars
   485→                      </p>
   486→                    </div>
   487→                  </Tabs>
   488→                  {descTab === "write" ? (
   489→                    <FormField
   490→                      control={form.control}
   491→                      name="description"
   492→                      render={({ field }) => (
   493→                        <FormItem className="mt-3">
   494→                          <FormControl>
   495→                            <Textarea
   496→                              {...field}
   497→                              value={field.value ?? ""}
   498→                              rows={12}
   499→                              placeholder={"An honest review — who it is for, who should skip it..."}
   500→                              className="min-h-[280px] font-mono text-[13px] leading-relaxed"
   501→                            />
   502→                          </FormControl>
   503→                          <FormMessage />
   504→                        </FormItem>
   505→                      )}
   506→                    />
   507→                  ) : (
   508→                    <div className="mt-3 min-h-[280px] rounded-lg border p-4">
   509→                      {(watched.description ?? "").trim() ? (
   510→                        <React.Suspense fallback={<MarkdownFallback />}>
   511→                          <MarkdownBlock content={watched.description ?? ""} />
   512→                        </React.Suspense>
   513→                      ) : (
   514→                        <p className="text-sm text-muted-foreground">Nothing to preview yet.</p>
   515→                      )}
   516→                    </div>
   517→                  )}
   518→                </div>
   519→
   520→                <FormField
   521→                  control={form.control}
   522→                  name="affiliateUrl"
   523→                  render={({ field }) => (
   524→                    <FormItem>
   525→                      <FormLabel>Affiliate URL</FormLabel>
   526→                      <FormControl>
   527→                        <div className="flex gap-2">
   528→                          <Input
   529→                            {...field}
   530→                            placeholder="https://www.amazon.in/dp/...?tag=mnkp-21"
   531→                            className="h-10 font-mono text-[13px]"
   532→                          />
   533→                          {/^https?:\/\//.test(field.value ?? "") ? (
   534→                            <a
   535→                              href={field.value ?? ""}
   536→                              target="_blank"
   537→                              rel="noopener noreferrer"
   538→                              aria-label="Open affiliate URL in a new tab"
   539→                              title="Open affiliate URL"
   540→                              className="flex size-10 shrink-0 items-center justify-center rounded-md border text-muted-foreground transition-colors hover:bg-muted hover:text-primary"
   541→                            >
   542→                              <ExternalLink className="size-4" aria-hidden="true" />
   543→                            </a>
   544→                          ) : null}
   545→                        </div>
   546→                      </FormControl>
   547→                      <FormDescription>Where the Buy button sends visitors</FormDescription>
   548→                      <FormMessage />
   549→                    </FormItem>
   550→                  )}
   551→                />
   552→              </CardContent>
   553→            </Card>
   554→
   555→            {/* Pricing */}
   556→            <Card>
   557→              <CardHeader className="pb-3">
   558→                <CardTitle className="text-base">Pricing</CardTitle>
   559→              </CardHeader>
   560→              <CardContent className="space-y-5">
   561→                <div className="grid gap-5 sm:grid-cols-2">
   562→                  <FormField
   563→                    control={form.control}
   564→                    name="price"
